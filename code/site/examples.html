Welcome to TB

A short tour of TB: variables, lists, loops, functions, maps and care().

Download welcome.tb
# welcome.tb
# A short tour of TB: one small program, many features.
# Run it with:  tb welcome.tb

# Variables and f-strings
let name = "Tanisha"
say(f"Hello, {name}! Welcome to TB.")
say("This program is a quick tour of the language.")
say("")

# Lists and loops
let languages = ["TB", "Python", "JavaScript", "Go"]
say("A few programming languages:")
let count = 1
for language in languages {
  say(str(count) + ". " + language)
  count = count + 1
}
say(f"That's {len(languages)} languages in one list.")
say("")

# Maps: named values that belong together
let pet = {"name": "Biscuit", "kind": "cat", "age": 3}
say(f"{pet.name} is a {pet.kind} who is {pet.age} years old.")
say("")

# A function that remembers things (a closure)
fn make_counter() {
  let steps = 0
  return fn() {
    steps = steps + 1
    return steps
  }
}

let walk = make_counter()
walk()
walk()
say("Steps taken so far: " + str(walk()))

# A function that calls itself (recursion)
fn stars(n) {
  if n <= 0 {
    return ""
  }
  return "* " + stars(n - 1)
}
say(stars(5))
say("")

# Counting down with while
let i = 3
while i > 0 {
  say(str(i) + "...")
  i = i - 1
}

if len(languages) > 3 {
  say("Lift-off! You have seen the basics of TB.")
} else {
  say("Lift-off!")
}

say("")
say("A note from TB:")
care(name)
say("")
say("Have fun writing your own programs.")

Compliment machine

Mixes words from two lists to write lots of little compliments.

Download compliments.tb
# Title: Compliment machine
# About: Mixes words from two lists to write lots of little compliments.

let things = ["idea", "effort", "kindness"]
let words = ["makes a difference", "is noticed"]

fn compliment(thing, word) {
  return "Your " + thing + " " + word + "."
}

# A loop inside a loop: every thing gets paired with every word.
for thing in things {
  for word in words {
    say(compliment(thing, word))
  }
}
say("")
say("That's " + str(len(things) * len(words)) + " compliments, and every one is true.")

Grocery list

Builds a shopping list with push(), changes an item, and prints it with for.

Download grocery_list.tb
# Title: Grocery list
# About: Builds a shopping list with push(), changes an item, and prints it with for.

let groceries = []
push(groceries, "milk")
push(groceries, "bread")
push(groceries, "strawberries")
push(groceries, "chocolate")

say("My list has " + str(len(groceries)) + " things:")
for item in groceries {
  say("- " + item)
}
say("")

# Lists start counting at 0, so groceries[0] is the first thing.
say("First thing to buy: " + groceries[0])

# Change an item in the list.
groceries[1] = "fresh bread"
say("Oops, make that: " + groceries[1])
say("")

# Two lists can be joined with +
let treats = ["ice cream", "flowers"]
let everything = groceries + treats
say("With treats added:", everything)

# Number each item, starting from 1 for humans.
let number = 1
for item in everything {
  say(str(number) + ". " + item)
  number = number + 1
}

Functions that remember

Closures: little functions that keep their own memory, like a step counter.

Download closures.tb
# Title: Functions that remember
# About: Closures: little functions that keep their own memory, like a step counter.

fn make_counter(label) {
  let count = 0
  return fn() {
    count = count + 1
    say(label + ": " + str(count))
  }
}

let cups = make_counter("Cups of tea today")
let pages = make_counter("Pages read today")

cups()
cups()
pages()
cups()

# A function can also be kept in a 'let' and passed around like any value.
let shout = fn(text) {
  return text + "!!!"
}

fn twice(f, value) {
  return f(f(value))
}

say(twice(shout, "Hello there"))

Functions that call themselves

Recursion: Fibonacci numbers, factorials, and a staircase of stars.

Download recursion.tb
# Title: Functions that call themselves
# About: Recursion: Fibonacci numbers, factorials, and a staircase of stars.

# Each Fibonacci number is the two before it added together.
fn fib(n) {
  if n < 2 {
    return n
  }
  return fib(n - 1) + fib(n - 2)
}

let numbers = []
for i in range(10) {
  push(numbers, fib(i))
}
say("The first 10 Fibonacci numbers:", numbers)

# 5 factorial means 5 * 4 * 3 * 2 * 1.
fn factorial(n) {
  if n <= 1 {
    return 1
  }
  return n * factorial(n - 1)
}
say("5 factorial is", factorial(5))

# Draw a staircase by calling stairs() with a smaller number each time.
fn stairs(n) {
  if n == 0 {
    return
  }
  stairs(n - 1)
  say("*" * n)
}
stairs(4)

Guess the number

A small guessing game. TB is thinking of a number from 1 to 20.

Download guess_the_number.tb
# Title: Guess the number
# About: A small guessing game. TB is thinking of a number from 1 to 20.

let secret = 7
let tries = 0
let found = false

say("I'm thinking of a number from 1 to 20. Can you guess it?")

while not found and tries < 5 {
  let answer = ask("Your guess: ")
  # Check the answer is a whole number from 1 to 20 before using num().
  let ok = false
  for n in range(1, 21) {
    if answer == str(n) {
      ok = true
    }
  }
  if not ok {
    say("Hmm, \"" + answer + "\" isn't a number from 1 to 20. Try again!")
  } else {
    tries = tries + 1
    let guess = num(answer)
    if guess == secret {
      found = true
    } else if guess < secret {
      say("A bit higher!")
    } else {
      say("A bit lower!")
    }
  }
}

if found {
  say("Yes! It was " + str(secret) + ". You got it in " + str(tries) + " tries.")
} else {
  say("Out of tries! The number was " + str(secret) + ". Want to play again?")
}

Care bot

A gentle little bot that asks how you are feeling and answers kindly.

Download care_bot.tb
# Title: Care bot
# About: A gentle little bot that asks how you are feeling and answers kindly.

# Checks whether a word is in a list of words.
fn is_one_of(word, options) {
  for option in options {
    if word == option {
      return true
    }
  }
  return false
}

say("Hi! I'm Care Bot. I'm here to check in on you.")
let name = ask("What's your name? ")
if name == "" {
  name = "friend"
}
say("It's good to see you, " + name + ".")
say("")

say("How are you feeling right now?")
say("  1. Happy")
say("  2. Tired")
say("  3. Sad")
say("  4. Stressed")
say("  5. Something else")
let feeling = ask("Type a number (or the word): ")
say("")

if is_one_of(feeling, ["1", "happy", "Happy", "HAPPY", "good", "Good", "great", "Great"]) {
  say("Yay! That makes me so happy to hear.")
  say("Hold on to this feeling. You deserve every bit of it.")
} else if is_one_of(feeling, ["2", "tired", "Tired", "TIRED", "sleepy", "Sleepy"]) {
  say("Sounds like you've been working hard.")
  say("Maybe a glass of water and a little rest? Even ten minutes helps.")
} else if is_one_of(feeling, ["3", "sad", "Sad", "SAD", "down", "Down"]) {
  say("I'm sorry today feels heavy. It's okay to feel sad.")
  say("Wrap up in something cosy, and maybe talk to someone you trust.")
} else if is_one_of(feeling, ["4", "stressed", "Stressed", "STRESSED", "worried", "Worried"]) {
  say("That sounds like a lot. Let's take one slow breath together.")
  say("In... and out. You don't have to do everything at once.")
} else {
  say("Thank you for telling me. Whatever you're feeling is okay.")
  say("Be gentle with yourself today.")
}
say("")

# Ask for a score, but only accept whole numbers from 1 to 10.
let answer = ask("On a scale of 1 to 10, how is your day going? ")
let score = 0
for n in range(1, 11) {
  if answer == str(n) {
    score = n
  }
}

if score == 0 {
  say("That's alright, numbers can be hard to pick. I'm still glad you're here.")
} else if score <= 3 {
  say(str(score) + " out of 10. I'm sorry it's been a hard day.")
  say("Please rest, drink some water, and reach out to someone you trust.")
} else if score <= 6 {
  say(str(score) + " out of 10. A middling day. Something small and nice might help.")
} else if score <= 9 {
  say(str(score) + " out of 10! That's a good day. Well done.")
} else {
  say("10 out of 10! Amazing! Your happiness is contagious.")
}

say("")
say("One last thing before you go:")
care(name)

Apps

Complete programs written in TB. Desktop apps open windows; the security and network tools run in a terminal. Every one starts from a readable main.tb.

They already come inside your TB download, in its apps folder (with sample logs in apps/data). On Windows, the installer also puts the desktop apps in the Start menu. To run one yourself:

Your TBRun the To-do app
Windows installerStart menu, TB, To-do, or in TB Prompt: tb "%LOCALAPPDATA%\Programs\TB\apps\todo\main.tb"
Windows ziptb\tbw.exe tb\apps\todo\main.tb
Mac~/tb/tb ~/tb/apps/todo/main.tb
Linux./tb/tb tb/apps/todo/main.tb

Command-line tools take a file to read, for example tb apps/log-triage/main.tb apps/data/auth.log. The source links below are for reading or copying a single app.
